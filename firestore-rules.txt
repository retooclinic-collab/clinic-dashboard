// ===== Firestore Security Rules for clinic-dashboard =====
// Firebase Console > Firestore > Rules 에 붙여넣기
// 2026-06-27 업데이트 (card_vendor_checks 추가 — 거래처 미확인 메모/확인상태)

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // 관리자 이메일
    function isAdmin() {
      return request.auth != null && request.auth.token.email == 'retooclinic@gmail.com';
    }

    // 인증된 사용자
    function isAuthenticated() {
      return request.auth != null;
    }

    // consultations 컬렉션
    match /consultations/{docId} {
      // 읽기: 관리자는 전체, 직원은 최근 30일만
      allow read: if isAdmin()
        || (isAuthenticated() && resource.data.serverTs > request.time - duration.value(30, 'd'));

      // 쓰기(생성): 인증된 사용자 모두 가능
      allow create: if isAuthenticated();

      // 수정: 인증된 사용자 (개별 수정/소프트삭제)
      allow update: if isAuthenticated();

      // 하드삭제: 관리자만
      allow delete: if isAdmin();
    }

    // draft_imports 컬렉션 (파싱 초안 — 확정 전 임시 데이터)
    // ★ 2026-06-26 복구: 카드/은행/재고 규칙 재작성 때 누락돼 초안 탭이 권한오류 났던 것
    match /draft_imports/{docId} {
      // 읽기/쓰기: 로그인한 사용자 (초안 파싱 워크플로우)
      allow read, write: if isAuthenticated();
    }

    // audit_log 컬렉션 (감사 로그)
    match /audit_log/{docId} {
      // 읽기: 관리자만
      allow read: if isAdmin();
      // 쓰기: 인증된 사용자 (로그 기록용)
      allow create: if isAuthenticated();
      // 수정/삭제: 불가
      allow update, delete: if false;
    }

    // clinic_archive 컬렉션 (월별 아카이브)
    match /clinic_archive/{docId} {
      // 읽기/쓰기: 관리자만
      allow read, write: if isAdmin();
    }

    // clinic_settings 컬렉션 (경영분석 설정: 재료비단가, 수수료율, 고정비)
    match /clinic_settings/{docId} {
      // 읽기: 관리자 전체, 직원은 staff_list만 (로그인 차단 확인용)
      allow read: if isAdmin() || (isAuthenticated() && docId == 'staff_list');
      // 쓰기: 관리자만
      allow write: if isAdmin();
    }

    // access_log 컬렉션 (접속 로그)
    match /access_log/{docId} {
      allow read: if isAdmin();
      allow create: if isAuthenticated();
      allow update, delete: if false;
    }

    // users 컬렉션 (직원 설정용, 향후 확장)
    match /users/{uid} {
      allow read: if isAuthenticated() && (request.auth.uid == uid || isAdmin());
      allow write: if isAdmin();
    }

    // ===== 카드 결제내역 (CODEF 자동수집, 서버=서비스계정만 쓰기) =====
    match /card_expenses/{doc} {
      allow read: if isAuthenticated();   // 로그인한 직원만 조회
      allow write: if false;              // 쓰기는 서버(서비스계정)만 → 외부 쓰기 차단
    }
    match /card_sync_log/{doc} {
      allow read: if isAuthenticated();
      allow write: if false;
    }
    // 거래처 미확인(PG 뭉뚱그림) 확인상태/메모 — 로그인 사용자 읽기/쓰기 (메모만 저장)
    match /card_vendor_checks/{doc} {
      allow read, write: if isAuthenticated();
    }

    // ===== 은행 입출금내역 (CODEF 자동수집, 서버=서비스계정만 쓰기) =====
    match /bank_transactions/{doc} {
      allow read: if isAuthenticated();   // 로그인한 직원만 조회
      allow write: if false;              // 쓰기는 서버(서비스계정)만 → 외부 쓰기 차단
    }
    match /bank_sync_log/{doc} {
      allow read: if isAuthenticated();
      allow write: if false;
    }

    // ===== 재고자산: 현재고 미러 (Apps Script가 소유자 OAuth로 씀 → 규칙 우회) =====
    match /inventory_stock/{doc} {
      allow read: if isAuthenticated();   // 로그인한 직원/원장 조회
      allow write: if false;              // 클라이언트 쓰기 차단(미러는 IAM 권한으로)
    }
    // ===== 재고자산: 일별 평가 스냅샷 (원장 브라우저가 기록) =====
    match /inventory_value_daily/{doc} {
      allow read: if isAdmin();
      allow write: if isAdmin();
    }
  }
    // ===== JRK(제이알케이) 재무 — 원장(admin) 전용, 서버(SA)만 쓰기 =====
    match /jrk_card_expenses/{doc}        { allow read: if isAdmin(); allow write: if false; }
    match /jrk_bank_transactions/{doc}    { allow read: if isAdmin(); allow write: if false; }
    match /jrk_card_sync_log/{doc}        { allow read: if isAdmin(); allow write: if false; }
    match /jrk_bank_sync_log/{doc}        { allow read: if isAdmin(); allow write: if false; }
    match /jrk_card_vendor_checks/{doc}   { allow read, write: if isAdmin(); }
    match /jrk_consultations/{doc}        { allow read, write: if isAdmin(); }
    match /jrk_clinic_archive/{doc}       { allow read, write: if isAdmin(); }
    match /jrk_clinic_settings/{doc}      { allow read, write: if isAdmin(); }
    match /jrk_draft_imports/{doc}        { allow read, write: if isAdmin(); }
    match /jrk_inventory_stock/{doc}      { allow read: if isAdmin(); allow write: if false; }
    match /jrk_inventory_value_daily/{doc}{ allow read, write: if isAdmin(); }
    match /jrk_access_log/{doc}           { allow read: if isAdmin(); allow create: if isAuthenticated(); allow update, delete: if false; }
    match /jrk_audit_log/{doc}            { allow read: if isAdmin(); allow create: if isAuthenticated(); allow update, delete: if false; }
}
